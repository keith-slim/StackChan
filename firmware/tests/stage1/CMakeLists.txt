cmake_minimum_required(VERSION 3.16)
project(stackchan_stage1_clock LANGUAGES CXX)
enable_testing()
add_executable(stage1_clock_test clock_test.cpp)
add_executable(stage1_weather_change_test weather_change_test.cpp)
foreach(target IN ITEMS stage1_clock_test stage1_weather_change_test)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_include_directories(${target} PRIVATE ../../main/stackchan/port_stage1)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  endif()
endforeach()
add_test(NAME stage1_clock_test COMMAND stage1_clock_test)
add_test(NAME stage1_weather_change_test COMMAND stage1_weather_change_test)
